// Package action is the platform's typed transactional Action kernel. A batch
// status transition is declared as a Transition; the kernel owns the database
// transaction, eligibility, audit, idempotency and reconciliation fingerprint.
// It is not a general workflow engine or a sandbox for untrusted extensions.
package action

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"sort"
	"strings"
	"unicode/utf8"

	"github.com/XUJiahua/enterprise_agent/internal/app"
	"github.com/XUJiahua/enterprise_agent/internal/ledger"
	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"
)

// Transition declares one batch Action. Operation is stored in ActionRequest
// and ApprovalAudit; Field names the single response field.
type Transition struct {
	Operation, From, To, Field string
}

// Approve is the existing batch approval Action.
var Approve = Transition{Operation: "bulk-approve.v3", From: "submitted", To: "approved", Field: "approvedIds"}

type Rejection struct{ Status int }

func (r *Rejection) Error() string { return fmt.Sprintf("action rejected: %d", r.Status) }

type Executor struct{ pool *pgxpool.Pool }

// New requires a guarded Purchase runtime and append-only Action metadata tables.
func New(ctx context.Context, pool *pgxpool.Pool) (*Executor, error) {
	if pool == nil {
		return nil, errors.New("missing pool")
	}
	var guarded bool
	err := pool.QueryRow(ctx, `SELECT c.relrowsecurity AND c.relforcerowsecurity AND NOT
 (r.rolsuper OR r.rolbypassrls OR pg_has_role(current_user,c.relowner,'MEMBER'))
 FROM pg_roles r CROSS JOIN pg_class c WHERE r.rolname=current_user AND c.oid='"Purchase"'::regclass`).Scan(&guarded)
	if err != nil || !guarded {
		return nil, fmt.Errorf("Action requires non-owner guarded Purchase runtime: %v", err)
	}
	var configured bool
	err = pool.QueryRow(ctx, `SELECT count(*)=2 AND bool_and(c.relrowsecurity AND c.relforcerowsecurity
	 AND NOT pg_has_role(current_user,c.relowner,'MEMBER')
	 AND has_table_privilege(current_user,c.oid,'SELECT')
	 AND has_table_privilege(current_user,c.oid,'INSERT')
	 AND NOT has_any_column_privilege(current_user,c.oid,'UPDATE')
	 AND NOT has_table_privilege(current_user,c.oid,'DELETE,TRUNCATE,TRIGGER'))
	 FROM pg_class c WHERE c.oid IN ('"ActionRequest"'::regclass,'"ApprovalAudit"'::regclass)`).Scan(&configured)
	if err != nil || !configured {
		return nil, fmt.Errorf("Action tables require non-owner RLS and append-only grants: %v", err)
	}
	return &Executor{pool: pool}, nil
}

func canonical(ids []string) ([]string, error) {
	if len(ids) < 1 || len(ids) > 100 {
		return nil, &Rejection{400}
	}
	result := append([]string(nil), ids...)
	sort.Strings(result)
	for i, id := range result {
		if len(id) == 0 || len(id) > 100 || !utf8.ValidString(id) || (i > 0 && result[i-1] == id) {
			return nil, &Rejection{400}
		}
	}
	return result, nil
}

func validKey(key string) bool {
	if len(key) < 1 || len(key) > 128 {
		return false
	}
	for _, c := range key {
		if !(c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z' || c >= '0' && c <= '9' || strings.ContainsRune("._:-", c)) {
			return false
		}
	}
	return true
}

// Run executes t for the authenticated approver and returns the sorted IDs.
// The idempotency scope is t.Operation+tenant+actor+key; a replay returns the
// stored IDs with replayed=true. Ineligible targets (missing, other tenant,
// own, not in t.From) are indistinguishable: Rejection{409}.
func (e *Executor) Run(ctx context.Context, u app.Identity, t Transition, key string, input []string) ([]string, bool, error) {
	if t.Operation == "" || t.From == "" || t.To == "" || t.Field == "" {
		return nil, false, errors.New("incomplete transition")
	}
	if u.Role != "approver" || u.User == "" || u.Tenant == "" {
		return nil, false, &Rejection{403}
	}
	ids, err := canonical(input)
	if err != nil {
		return nil, false, err
	}
	if !validKey(key) {
		return nil, false, &Rejection{400}
	}
	for _, id := range ids {
		if strings.ContainsRune(id, 0) {
			return nil, false, &Rejection{404}
		}
	}
	tx, err := e.pool.Begin(ctx)
	if err != nil {
		return nil, false, err
	}
	defer func() { _ = tx.Rollback(ctx) }()
	if _, err = tx.Exec(ctx, `SELECT set_config('app.tenant',$1,true),set_config('app.user',$2,true),set_config('app.role',$3,true)`, u.Tenant, u.User, u.Role); err != nil {
		return nil, false, err
	}
	lock, _ := json.Marshal([]string{t.Operation, u.Tenant, u.User, key})
	if _, err = tx.Exec(ctx, `SELECT pg_advisory_xact_lock(hashtextextended($1,0))`, string(lock)); err != nil {
		return nil, false, err
	}
	hash := ledger.Fingerprint(t.Operation, ids)
	var savedHash string
	var saved []byte
	err = tx.QueryRow(ctx, `SELECT "requestHash",response FROM "ActionRequest"
	 WHERE "tenantId"=$1 AND "actorId"=$2 AND operation=$3 AND key=$4`, u.Tenant, u.User, t.Operation, key).Scan(&savedHash, &saved)
	if err == nil {
		if savedHash != hash {
			return nil, false, &Rejection{409}
		}
		var body map[string][]string
		if err = json.Unmarshal(saved, &body); err != nil {
			return nil, false, err
		}
		if err = tx.Commit(ctx); err != nil {
			return nil, false, err
		}
		return body[t.Field], true, nil
	}
	if !errors.Is(err, pgx.ErrNoRows) {
		return nil, false, err
	}
	rows, err := tx.Query(ctx, `SELECT id FROM "Purchase" WHERE id=ANY($1::text[])
	 AND "tenantId"=$2 AND "createdBy"<>$3 AND status=$4 ORDER BY id FOR UPDATE`, ids, u.Tenant, u.User, t.From)
	if err != nil {
		return nil, false, err
	}
	eligible := []string{}
	for rows.Next() {
		var id string
		if err = rows.Scan(&id); err != nil {
			rows.Close()
			return nil, false, err
		}
		eligible = append(eligible, id)
	}
	err = rows.Err()
	rows.Close()
	if err != nil {
		return nil, false, err
	}
	if len(eligible) != len(ids) {
		return nil, false, &Rejection{409}
	}
	tag, err := tx.Exec(ctx, `UPDATE "Purchase" SET status=$4 WHERE id=ANY($1::text[])
	 AND "tenantId"=$2 AND "createdBy"<>$3 AND status=$5`, ids, u.Tenant, u.User, t.To, t.From)
	if err != nil {
		return nil, false, err
	}
	if tag.RowsAffected() != int64(len(ids)) {
		return nil, false, &Rejection{409}
	}
	_, err = tx.Exec(ctx, `INSERT INTO "ApprovalAudit" ("tenantId","purchaseId","approvedBy",operation,"idempotencyKey","fromStatus","toStatus")
	 SELECT $2,id,$3,$4,$5,$6,$7 FROM unnest($1::text[]) AS id`, ids, u.Tenant, u.User, t.Operation, key, t.From, t.To)
	if err != nil {
		return nil, false, err
	}
	response, err := json.Marshal(map[string][]string{t.Field: ids})
	if err != nil {
		return nil, false, err
	}
	_, err = tx.Exec(ctx, `INSERT INTO "ActionRequest" ("tenantId","actorId",operation,key,"requestHash",response)
	 VALUES ($1,$2,$3,$4,$5,$6::jsonb)`, u.Tenant, u.User, t.Operation, key, hash, string(response))
	if err != nil {
		return nil, false, err
	}
	if err = tx.Commit(ctx); err != nil {
		return nil, false, err
	}
	return ids, false, nil
}
