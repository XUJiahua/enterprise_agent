package ledger

import "context"

type Drifted struct{ TenantID, PurchaseID, Status, Audited string }

// Drift returns every Purchase whose status is not the toStatus of its latest
// ApprovalAudit row; a purchase without audit rows must still be submitted.
func Drift(ctx context.Context, q Querier) ([]Drifted, error) {
	rows, err := q.Query(ctx, `SELECT p."tenantId",p.id,p.status,COALESCE(a."toStatus",'submitted')
	 FROM "Purchase" p LEFT JOIN LATERAL (SELECT "toStatus" FROM "ApprovalAudit" a
	  WHERE a."tenantId"=p."tenantId" AND a."purchaseId"=p.id ORDER BY a."createdAt" DESC LIMIT 1) a ON true
	 WHERE p.status <> COALESCE(a."toStatus",'submitted') ORDER BY 1,2`)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var drifted []Drifted
	for rows.Next() {
		var d Drifted
		if err = rows.Scan(&d.TenantID, &d.PurchaseID, &d.Status, &d.Audited); err != nil {
			return nil, err
		}
		drifted = append(drifted, d)
	}
	return drifted, rows.Err()
}
